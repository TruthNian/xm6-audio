cmake_minimum_required(VERSION 3.21)
project(XM6PcmBenchmark LANGUAGES C CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This computational benchmark uses Windows timing APIs")
endif()
set(repo "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(XM6_PCM_SOURCE "${repo}/core/src/pcm_ring.c" CACHE FILEPATH "PCM source to measure")
set(XM6_PCM_LABEL current CACHE STRING "Label for CSV output")
set(XM6_PCM_BASELINE "" CACHE FILEPATH "Optional baseline source for paired timing in one process")
if(NOT XM6_PCM_LABEL MATCHES "^[a-zA-Z0-9_-]+$")
    message(FATAL_ERROR "Use an identifier for XM6_PCM_LABEL")
endif()
add_executable(pcm_benchmark main.cpp)
target_compile_features(pcm_benchmark PRIVATE cxx_std_17)
target_include_directories(pcm_benchmark PRIVATE "${repo}/core/include")
target_compile_definitions(pcm_benchmark PRIVATE BENCH_VARIANT="${XM6_PCM_LABEL}")
set_property(TARGET pcm_benchmark PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
if(XM6_PCM_BASELINE)
    file(READ "${repo}/core/include/oa2dp/pcm_ring.h" header)
    string(REGEX MATCHALL "oa2dp_[a-zA-Z0-9_]+[ \t\r\n]*\\(" declarations "${header}")
    set(functions)
    foreach(declaration IN LISTS declarations)
        string(REGEX REPLACE "[ \t\r\n]*\\($" "" function "${declaration}")
        list(APPEND functions "${function}")
    endforeach()
    list(REMOVE_DUPLICATES functions)
    if(NOT "oa2dp_pcm_prepare_encoder" IN_LIST functions)
        message(FATAL_ERROR "Cannot identify the PCM API for paired timing")
    endif()
    foreach(variant before after)
        if(variant STREQUAL before)
            set(source "${XM6_PCM_BASELINE}")
        else()
            set(source "${XM6_PCM_SOURCE}")
        endif()
        add_library(pcm_${variant} OBJECT "${source}")
        target_include_directories(pcm_${variant} PRIVATE "${repo}/core/include")
        set_property(TARGET pcm_${variant} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
        foreach(function IN LISTS functions)
            target_compile_definitions(pcm_${variant} PRIVATE "${function}=${variant}_${function}")
        endforeach()
        target_sources(pcm_benchmark PRIVATE $<TARGET_OBJECTS:pcm_${variant}>)
        if(MSVC)
            set_source_files_properties("${source}" PROPERTIES COMPILE_OPTIONS
                "/FAs;/Fa${CMAKE_CURRENT_BINARY_DIR}/pcm-${variant}.asm")
        endif()
    endforeach()
    target_compile_definitions(pcm_benchmark PRIVATE BENCH_PAIRED)
else()
    target_sources(pcm_benchmark PRIVATE "${XM6_PCM_SOURCE}")
endif()
if(MSVC AND NOT XM6_PCM_BASELINE)
    set_source_files_properties("${XM6_PCM_SOURCE}" PROPERTIES COMPILE_OPTIONS
        "/FAs;/Fa${CMAKE_CURRENT_BINARY_DIR}/pcm.asm")
endif()
